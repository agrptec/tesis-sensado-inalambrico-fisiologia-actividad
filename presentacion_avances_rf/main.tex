\vspace{0.22cm}
\begin{block}{Tesis metodológica}
Una mejora en potencia, magnitud o dispersión de retardos no garantiza que el
gemelo conserve la estructura compleja necesaria para sensado. La siguiente
validación debe medir la respuesta del canal ante una perturbación conocida.
\end{block}
\begin{alertblock}{Alcance de la evidencia actual}
Los resultados disponibles demuestran alineación compleja estática parcialmente
transferible dentro de dc01. Todavía no demuestran recuperación de respiración.
\end{alertblock}
\end{frame}

\begin{frame}{Punto de partida: sensibilidad de fase a geometría}
\small
\begin{columns}[T]
\column{0.52\linewidth}
\begin{block}{Canal por trayectorias}
\[
H(f)=\sum_{\ell=1}^{L}\alpha_\ell(f)e^{-j2\pi f\tau_\ell}.
\]
Para una trayectoria, $\Delta\tau=\Delta d/c$ produce
\[
\Delta\phi_\ell=-2\pi f_c\Delta\tau_\ell
=-2\pi\frac{\Delta d_\ell}{\lambda}.
\]
\end{block}
\begin{alertblock}{Implicación}
Una escena visualmente precisa puede fallar en interferencia coherente por errores
sub-longitud de onda.
\end{alertblock}
\column{0.46\linewidth}
\centering
\includegraphics[width=\linewidth,height=0.62\textheight,keepaspectratio]{raw10_phase_sensitivity.pdf}
\end{columns}
\source{La escala $1/B$ describe resolubilidad; no impide estimar un máximo sub-bin bajo un modelo y SNR suficientes.}
\end{frame}

\begin{frame}{S4.1: por qué no asignar de inmediato una fase a cada camino}
\small
\begin{columns}[T]
\column{0.50\linewidth}
\begin{block}{Modelo local}
\[
H^{\rm mod}(f)=\sum_p\alpha_p(f)
e^{-j2\pi f\tau_p+j\varepsilon_p},
\]
\[
p(\varepsilon_p|\mu_p,\kappa_p)=
\frac{e^{\kappa_p\cos(\varepsilon_p-\mu_p)}}{2\pi I_0(\kappa_p)}.
\]
\end{block}
\column{0.48\linewidth}
\begin{block}{Estrategias contrastadas}
\begin{itemize}\setlength\itemsep{0.10cm}
\item \textbf{PEOC}: confía en la fase RT; puede sesgar materiales.
\item \textbf{UPEC}: marginaliza fase uniforme; estable, pero descarta estructura.
\item \textbf{PEAC}: incertidumbre von Mises; precisa, pero sensible a inicialización.
\end{itemize}
\end{block}
\end{columns}
\begin{alertblock}{Decisión}
Buscar primero estructura agregada de baja dimensión. La auditoría pathwise posterior
encontró coherencias casi unitarias y condición de hasta $1.45\times10^6$.
\end{alertblock}
\end{frame}

\begin{frame}{Derivación I: residuo conjugado}
\small
\begin{block}{Hipótesis afín mínima}
\[
\Hreal_k\approx c\,e^{-j2\pi\nu_k\Delta\tau}\Hrt_k+e_k,
\qquad c=|c|e^{j\phi_0},\quad \nu_k=f_k-f_{\rm ref}.
\]
\end{block}
Multiplicando por el conjugado del gemelo:
\[
R_k=\Hreal_k(\Hrt_k)^*
\approx c\,e^{-j2\pi\nu_k\Delta\tau}|\Hrt_k|^2.
\]
Como $|\Hrt_k|^2$ es real no negativo,
\begin{equation*}
\boxed{\angle R_k\approx\phi_0-2\pi\nu_k\Delta\tau.}
\end{equation*}
\begin{alertblock}{Por qué producto y no cociente}
$\Hreal/\Hrt$ es inestable cerca de nulos simulados. El producto conserva la
diferencia angular y pondera tonos con energía en ambos canales.
\end{alertblock}
\end{frame}

% 18 ----------------------------------------------------------------
\begin{frame}{Derivación II: frecuencia relativa, intercepto y residuo}
\small
\[
-2\pi f_k\Delta\tau=
-2\pi f_{\rm ref}\Delta\tau-2\pi\nu_k\Delta\tau.
\]
El primer término es constante en frecuencia y se incorpora al intercepto. En el
canal multitrayectoria la relación no es exacta, por lo que se escribe
\[
\boxed{
\Delta\phi_{i,u,a,k}\approx
\phi_{0,i,u,a}-2\pi\nu_k\Delta\tau_{{\rm eff},i,u}+r_{i,u,a,k}
\pmod{2\pi}.}
\]
\begin{columns}[T]
\column{0.31\linewidth}
\begin{block}{Orden cero}
$\phi_0$: rotación constante o CPO.
\end{block}
\column{0.31\linewidth}
\begin{block}{Primer orden}
$\dtau$: pendiente fase--frecuencia.
\end{block}
\column{0.31\linewidth}
\begin{block}{Orden superior}
$r$: curvatura y discrepancia no afín.
\end{block}
\end{columns}
\end{frame}

\begin{frame}{S4.2: detección inicial de una pendiente compartida}
\small
Al formar incrementos adyacentes, el intercepto desaparece:
\[
R_{k+1}R_k^*\approx B_{k+1}B_k
e^{-j2\pi\Delta f\dtau},\qquad
\angle(R_{k+1}R_k^*)\approx-2\pi\Delta f\dtau.
\]
\vspace{0.12cm}

\begin{tabular}{lrr}
\toprule

\end{alertblock}
\end{frame}

\begin{frame}{S4.2: retardo oráculo como techo de alineabilidad}
\small
\begin{equation*}
\widehat\tau^{\rm oracle}_{i,u}=
\arg\max_{\tau\in[-500,500]\,{\rm ns}}
\left|\sum_a\sum_k R_{i,u,a,k}e^{j2\pi\nu_k\tau}\right|.
\end{equation*}

\end{alertblock}

\end{exampleblock}

\begin{frame}{Random Forest: modelo y operación del conjunto}
\small
Para cada árbol $b=1,\ldots,T$ se forma una muestra bootstrap y se crece un
regresor $h_b$. En cada nodo se selecciona una variable y un umbral que reducen el
error cuadrático:
\[
(j^*,s^*)=\arg\min_{j,s}
\left[
\sum_{n\in\mathcal I_L(j,s)}(y_n-\bar y_L)^2+
\sum_{n\in\mathcal I_R(j,s)}(y_n-\bar y_R)^2
\right].
\]
La predicción es el promedio de los árboles:
\[
\boxed{\widehat\tau^{\rm pred}(\vect x)=
\frac1T\sum_{b=1}^{T}h_b(\vect x).}
\]
\begin{columns}[T]
\column{0.48\linewidth}
\begin{block}{Por qué es útil aquí}
Captura interacciones no lineales entre posición y descriptores RT sin imponer una
forma global para el mapa de retardo.
\end{block}
\column{0.48\linewidth}
\begin{alertblock}{Lo que no identifica}
No separa STO, geometría, cables o trayectorias ausentes. Predice el escalar efectivo
definido por el oráculo.
\end{alertblock}
\end{columns}
\source{Se omiten valores de hiperparámetros que no están documentados en los artefactos proporcionados.}
\end{frame}

\begin{frame}{Por qué CPO y coherencia responden preguntas distintas}
\small
\begin{block}{Coherencia compleja}
\[
\rho(\vect h,\widehat{\vect h})=
\frac{|\vect h^H\widehat{\vect h}|}{\|\vect h\|_2\|\widehat{\vect h}\|_2}.
\]
Para cualquier $\psi$, $\rho(\vect h,e^{j\psi}\widehat{\vect h})=\rho(\vect h,\widehat{\vect h})$.
\end{block}
\begin{table}
\centering
\begin{tabular}{lrr}
\toprule
Tratamiento B0 denso & Error angular [$^\circ$] & Coherencia\\
\midrule
Bruto & 89.93 & 0.156\\
Sólo CPO & 81.42 & 0.156\\
Retardo predicho + CPO & 45.59 & \textbf{0.634}\\
Retardo oráculo + CPO & 31.63 & \textbf{0.745}\\
\bottomrule
\end{tabular}
\end{table}
\begin{alertblock}{Conclusión}
CPO cambia la referencia angular. El retardo rota cada frecuencia de forma distinta
y por ello modifica la geometría del vector multifrecuencia.
\end{alertblock}
\end{frame}

\[
\phi_{0,i,u,a}\approx\beta_{u,a}+\psi_{i,u}+\epsilon_{i,u,a}\pmod{2\pi}.
\]
\begin{columns}[T]

\begin{block}{$\beta_{u,a}$}

\end{block}
\column{0.32\linewidth}
\begin{block}{$\psi_{i,u}$}

\end{block}
\column{0.32\linewidth}

\end{alertblock}
\end{columns}

\centering\scriptsize No se concluye aleatoriedad universal: pueden faltar estado de PLL, trigger, CFO/SFO o metadatos de adquisición.
\end{frame}

\begin{frame}{Factorización final: tres roles inferenciales}
\small
\begin{equation*}
\boxed{
\Delta\phi_{i,u,a,k}\approx
-2\pi\nu_k\Delta\tau_{{\rm eff},i,u}+\psi_{i,u}+r^{\rm rel}_{i,u,a,k}
\pmod{2\pi}.}
\end{equation*}
\begin{table}
\centering
\scriptsize
\begin{tabularx}{\linewidth}{p{2.2cm}YY}
\toprule
Componente & Evidencia & Decisión metodológica\\
\midrule
$\dtau$ & Transferencia parcial entre modelos, antenas, frecuencias y regiones &
Predecir con covariables disponibles y validar fuera de muestra.\\
$\psi$ & Dirección común dentro de captura, sin predictibilidad local demostrada &
Referenciar, medir o marginalizar como nuisance bajo el protocolo actual.\\
$r^{\rm rel}$ & Suavidad espectral local y asociación con distribución energética &
Modelar con estructura; evitar una fase libre por trayectoria no resoluble.\\
\bottomrule
\end{tabularx}
\end{table}
\begin{alertblock}{Cautela causal}
Los términos organizan grados de libertad; no prueban tres mecanismos físicos independientes.
\end{alertblock}
\end{frame}

\begin{frame}{La propiedad relevante para sensado es diferencial}
\small
\begin{columns}[T]
\column{0.49\linewidth}
\begin{block}{Fidelidad estática}
\[
\Hrt(f,x_0)\approx\Hreal(f,x_0).
\]
Describe si el gemelo reproduce el canal en una configuración fija.
\end{block}
\begin{exampleblock}{Fidelidad diferencial}
\[
J_x(f)=\frac{\partial H(f,x)}{\partial x},\qquad
\frac{\partial\Hrt}{\partial x}\approx
\frac{\partial\Hreal}{\partial x}.
\]
Determina si reproduce la perturbación causada por movimiento.
\end{exampleblock}
\column{0.49\linewidth}
\centering
\includegraphics[width=\linewidth,height=0.57\textheight,keepaspectratio]{raw11_differential_fidelity.pdf}
\begin{alertblock}{Consecuencia}
Coincidir en un punto no garantiza coincidir en la dirección del cambio.
\end{alertblock}
\end{columns}
\end{frame}

\begin{frame}{Amplitud y fase observan proyecciones distintas}
\small
Sea un canal local $H(x)=H_s+a e^{-j\kappa x}$. Entonces
\[
\frac{\partial H}{\partial x}=-j\kappa a e^{-j\kappa x},
\]
\[
\frac{\partial |H|}{\partial x}=
\frac{\operatorname{Re}\{H^*\partial H/\partial x\}}{|H|},
\qquad
\frac{\partial\angle H}{\partial x}=
\frac{\operatorname{Im}\{H^*\partial H/\partial x\}}{|H|^2}.
\]
\begin{columns}[T]
\column{0.48\linewidth}
\begin{block}{Dependencia geométrica}
Según la relación entre $H_s$ y la componente dinámica, amplitud o fase pueden ser
sensibles, o ambas pueden quedar cerca de un punto ciego.
\end{block}
\column{0.48\linewidth}
\begin{alertblock}{Cerca de un nulo}
La fase puede cambiar mucho por ruido. La selección debe ponderar sensibilidad con
incertidumbre; ``mayor variación de fase'' no basta.
\end{alertblock}
\end{columns}
\end{frame}
